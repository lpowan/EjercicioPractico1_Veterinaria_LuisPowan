!======================================================================
! Multilayer Switch0  ->  hostname SwitchMulticapaDistribucion
!======================================================================
enable
configure terminal
!--- 1. Configuracion basica
hostname SwitchMulticapaDistribucion
no ip domain-lookup
banner motd #Estas conectandote al Switch multicapa de distribucion#
enable secret CLASS
line console 0
 password CISCO
 login
 exit
line vty 0 15
 password CISCO
 login
 exit
service password-encryption
!--- 2. VLANs
vlan 10
 name Clientes
 exit
vlan 20
 name Administrativo
 exit
vlan 30
 name Finanzas
 exit
!--- 3. Interfaces
ip routing
interface FastEthernet0/24
 description Troncal 802.1Q hacia switch Administrativo (VLAN 10,20,30)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30
 no shutdown
 exit
interface GigabitEthernet0/1
 description Enlace IV capa 3 hacia RouterServidores (Router0 Gig0/1)
 no switchport
 ip address 172.16.0.14 255.255.255.252
 no shutdown
 exit
interface GigabitEthernet0/2
 description Enlace I capa 3 hacia RouterRedundancia2 (Router1 Gig0/2)
 no switchport
 ip address 172.16.0.2 255.255.255.252
 no shutdown
 exit
interface vlan 10
 description Gateway VLAN 10 Clientes + relay DHCP
 ip address 192.168.10.1 255.255.255.0
 ip helper-address 10.10.1.3
 no shutdown
 exit
interface vlan 20
 description Gateway VLAN 20 Administrativo + relay DHCP
 ip address 192.168.20.1 255.255.255.0
 ip helper-address 10.10.1.3
 no shutdown
 exit
interface vlan 30
 description Gateway VLAN 30 Finanzas + relay DHCP
 ip address 192.168.30.1 255.255.255.0
 ip helper-address 10.10.1.3
 no shutdown
 exit
!--- 4. Enrutamiento estatico (principal + flotante AD 10)
ip route 10.10.1.0 255.255.255.248 172.16.0.13
ip route 192.168.1.0 255.255.255.248 172.16.0.1
ip route 10.10.1.0 255.255.255.248 172.16.0.1 10
ip route 192.168.1.0 255.255.255.248 172.16.0.13 10
end
write memory
